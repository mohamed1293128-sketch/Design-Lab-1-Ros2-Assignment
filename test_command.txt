# Test Commands

# Build the workspace (run from the workspace root)
colcon build --symlink-install

# Source the workspace (run in every new terminal)
source install/setup.bash

# Start Turtlesim
ros2 run turtlesim turtlesim_node

# Start the pose control server (provides /set_speed and /go_to_pose)
ros2 run my_packages pose_control

# Start the reset service
ros2 run my_packages reset

# Start the square drawing action server
ros2 run my_packages square_draw

# Set the turtle's target pose (requires pose_control to be running)
ros2 run my_packages pose_set --ros-args -p x:=3.0 -p y:=4.0 -p yaw:=0.0

# Draw a square with side length 2.0 (requires square_draw and its servers)
ros2 run my_packages square_client --ros-args -p side_length:=2.0

# Change linear and angular speed
ros2 service call /set_speed interfaces/srv/Speed "{linear_velocity: 3.0, angular_velocity: 1.0}"

# Test invalid speed (should be rejected)
ros2 service call /set_speed interfaces/srv/Speed "{linear_velocity: 0.0, angular_velocity: 1.0}"

# Reset the turtle and clear the screen
ros2 service call /reset_node std_srvs/srv/Trigger "{}"

# Send a target pose directly (x, y in turtlesim; yaw in radians)
ros2 action send_goal /go_to_pose interfaces/action/GoToPose "{target_pose: {x: 1.0, y: 3.0, yaw: 1.57}}" --feedback

# Draw a square directly through the action
ros2 action send_goal /draw_square interfaces/action/DrawSquare "{side_length: 2.0}" --feedback

# Start the full task using the launch file instead of starting servers individually
ros2 launch bringup Task.launch.xml

# Run the pose client with values from pose.yaml
ros2 launch bringup pose_set.launch.xml

# Run the square client with side length from sidelegnth.yaml
ros2 launch bringup draw_square_client.launch.xml

# List active nodes
ros2 node list

# List available services
ros2 service list

# List available actions
ros2 action list
